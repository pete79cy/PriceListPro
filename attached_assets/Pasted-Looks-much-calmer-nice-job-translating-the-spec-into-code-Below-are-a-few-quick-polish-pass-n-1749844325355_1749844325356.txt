### Quick JS Hook for Bulk Bar

```js
const bulkBar = document.getElementById('bulk-bar');
function syncBulkBar() {
  const anyChecked = [...document.querySelectorAll('.row-cb')].some(cb => cb.checked);
  bulkBar.classList.toggle('show', anyChecked);
}
document.body.addEventListener('change', e => {
  if (e.target.classList.contains('row-cb') || e.target.id === 'master-cb') {
    syncBulkBar();
  }
});
```
