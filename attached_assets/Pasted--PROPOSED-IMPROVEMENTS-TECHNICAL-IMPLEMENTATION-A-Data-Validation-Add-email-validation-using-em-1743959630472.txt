A. Data Validation

Add email validation using email_validator.

from email_validator import validate_email, EmailNotValidError

def is_valid_email(email):
    if not email:
        return True  # Optional field
    try:
        validate_email(email)
        return True
    except EmailNotValidError:
        return False

Use this in POST routes before committing data.

    contact_date = db.Column(db.DateTime, default=datetime.utcnow)
    contact_type = db.Column(db.String(50))  # e.g. phone, email

D. Customer Search API

from sqlalchemy import or_

@app.route('/api/customers/search')
def search_customers():
    query = request.args.get('q', '')
    customers = Customer.query.filter(
        or_(
            Customer.name.ilike(f"%{query}%"),
            Customer.email.ilike(f"%{query}%"),
            Customer.phone.ilike(f"%{query}%")
        )
    ).all()
    return jsonify([{
        'id': c.id,
        'name': c.name,
        'email': c.email
    } for c in customers])

E. Customer Statistics Utility

def get_customer_stats(customer_id):
    customer = Customer.query.get(customer_id)
    if not customer:
        return {}

    invoices = customer.invoices
    return {
        'total_invoices': len(invoices),
        'total_price_lists': len(customer.price_lists),
        'last_order_date': max([i.invoice_date for i in invoices]) if invoices else None,
        'average_order_value': (
            sum([i.total_amount for i in invoices]) / len(invoices)
        ) if invoices else 0
    }
