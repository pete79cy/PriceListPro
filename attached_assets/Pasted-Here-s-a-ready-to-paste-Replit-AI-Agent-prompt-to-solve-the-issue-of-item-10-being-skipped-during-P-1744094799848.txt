import os
from jinja2 import Environment, FileSystemLoader
from weasyprint import HTML

def generate_quotation_pdf(quotation, upload_folder, use_modern_template=False, debug=False):
    env = Environment(loader=FileSystemLoader('templates/pdf'))
    template_name = 'modern_quotation_template.html' if use_modern_template else 'quotation_template.html'
    template = env.get_template(template_name)

    print(f"\nGenerating PDF for Quotation #{quotation.quotation_number}")
    for i, item in enumerate(quotation.items, 1):
        print(f"Item {i}: {item.description} | Qty: {item.quantity}")

    debug_styles = """
    <style>
      table {
        width: 100%;
        border-collapse: collapse;
        font-size: 12px;
        page-break-inside: auto;
      }
      thead { display: table-header-group; }
      tfoot { display: table-footer-group; }
      tr { page-break-inside: avoid; page-break-after: auto; }
      th, td {
        border: 1px solid black;
        padding: 6px;
        text-align: left;
        vertical-align: top;
      }
      tr.debug-row-10 {
        background-color: yellow;
        border: 2px solid red;
      }
    </style>
    """ if debug else ""

    rendered_html = template.render(
        quotation=quotation,
        debug_styles=debug_styles,
    )

    output_filename = f"{quotation.quotation_number}_quotation.pdf"
    pdf_file = os.path.join(upload_folder, output_filename)

    HTML(string=rendered_html).write_pdf(pdf_file)
    print(f"PDF saved to: {pdf_file}")
    return pdf_file
