**Phase 3: JavaScript for Interactivity (using jQuery)**

Create/update your JavaScript file (e.g., `static/js/customer_list.js`) and ensure it's loaded in your template.

```javascript
$(document).ready(function() {
    const $table = $('#customerTable');
    const $tbody = $table.find('tbody');
    const $headers = $table.find('thead th.sortable');
    const $filterButtons = $('.filter-btn');
    const $clearFiltersButton = $('#clearFilters');
    const $searchCustomersInput = $('#searchCustomers');
    const originalRows = $tbody.find('tr').toArray(); // Store original rows for clearing filters

    // --- Client-Side Sorting ---
    $headers.on('click', function() {
        const $header = $(this);
        const column = $header.data('column');
        let direction = $header.data('direction') || 'asc';

        // Toggle direction
        direction = (direction === 'asc') ? 'desc' : 'asc';
        $header.data('direction', direction);

        // Reset other headers' indicators and direction
        $headers.not($header).data('direction', null).find('.sort-indicator').removeClass('asc desc').text('');
        $header.find('.sort-indicator').removeClass('asc desc').addClass(direction).text(direction === 'asc' ? '▲' : '▼');

        // Get current rows from tbody
        let rows = $tbody.find('tr').toArray();

        rows.sort(function(a, b) {
            const valA = $(a).find('td').eq($header.index()).text().trim().toLowerCase();
            const valB = $(b).find('td').eq($header.index()).text().trim().toLowerCase();

            // Basic string comparison, can be enhanced for numbers/dates
            if (valA < valB) {
                return direction === 'asc' ? -1 : 1;
            }
            if (valA > valB) {
                return direction === 'asc' ? 1 : -1;
            }
            return 0;
        });

        $tbody.empty().append(rows);
    });

    // --- Delete Confirmation ---
    $tbody.on('click', '.delete-btn', function(e) {
        e.preventDefault();
        const customerId = $(this).data('id');
        const customerName = $(this).data('name');
        const deleteUrl = `/customer/delete/${customerId}`; // Adjust if your URL structure is different

        if (confirm(`Are you sure you want to delete customer "${customerName}" (ID: ${customerId})? This action cannot be undone.`)) {
            // Perform AJAX delete or redirect
            // For simplicity, redirecting. For AJAX, you'd make a $.post or $.ajax call.
            // window.location.href = deleteUrl; // If your delete route handles GET for confirmation (not recommended for actual delete)
            // Or use a form submission via POST
            const form = $('<form>', {
                'method': 'POST',
                'action': deleteUrl // This should be the URL for your Flask delete route
            }).append($('<input>', { // CSRF token if you're using Flask-WTF
                'name': 'csrf_token', // Adjust name if different
                'value': '{{ csrf_token() if csrf_token else "" }}', // Get CSRF token if available
                'type': 'hidden'
            }));
            $('body').append(form);
            form.submit();
        }
    });

    // --- Filter Management ---
    let activeFilters = {}; // To store current active filters

    function applyFiltersAndSearch() {
        const searchTerm = $searchCustomersInput.val().toLowerCase();
        $tbody.empty(); // Clear current rows

        let filteredRows = originalRows.filter(function(row) {
            const $row = $(row);
            let passSearch = true;
            let passFilters = true;

            // Search functionality
            if (searchTerm) {
                const rowText = $row.text().toLowerCase();
                if (rowText.indexOf(searchTerm) === -1) {
                    passSearch = false;
                }
            }

            // Filter functionality (example for 'missing-email')
            if (activeFilters['missing-email']) {
                const emailText = $row.find('td').eq($headers.filter('[data-column="email"]').index()).text().trim();
                if (emailText.toLowerCase() !== 'missing' && emailText !== 'n/a') { // Check against your "missing" text
                    passFilters = false;
                }
            }
            // Example for 'price-list' (you'll need to define how this is determined, e.g., a data attribute on the row)
            if (activeFilters['price-list']) {
                 // Assuming a customer object has a 'has_price_list' boolean property
                 // This requires that info to be available, e.g. as a data attribute on the <tr>
                 // For this example, let's say a 'data-has-price-list="true"' attribute exists on relevant rows
                if ($row.data('has-price-list') !== true) {
                    passFilters = false;
                }
            }
            // Example for categories (assuming category text is in the category column)
            if (activeFilters.category) {
                const categoryText = $row.find('td').eq($headers.filter('[data-column="category"]').index()).text().trim().toLowerCase();
                if (categoryText !== activeFilters.category) {
                    passFilters = false;
                }
            }
            // Add more filter logic here as needed

            return passSearch && passFilters;
        });

        $tbody.append(filteredRows);
        $('#customerCount').text(`Customers ${filteredRows.length}`);

        // Show/hide "Clear All Filters" button
        if (Object.keys(activeFilters).length > 0 || searchTerm) {
            $clearFiltersButton.show();
        } else {
            $clearFiltersButton.hide();
        }
    }

    // Search input
    $searchCustomersInput.on('keyup', debounce(applyFiltersAndSearch, 300));

    // Filter buttons
    $filterButtons.on('click', function() {
        const $button = $(this);
        const filterName = $button.data('filter'); // e.g., "missing-email", "price-list"
        
        if ($button.hasClass('categories-btn')) return; // Handle categories dropdown separately

        if ($button.hasClass('active')) {
            $button.removeClass('active');
            delete activeFilters[filterName];
        } else {
            $button.addClass('active');
            activeFilters[filterName] = true;
        }
        applyFiltersAndSearch();
    });

    // Categories Dropdown Logic (Simple Example)
    $('.categories-btn').on('click', function() {
        $(this).siblings('.dropdown-content').toggle();
    });

    $('.dropdown-content a').on('click', function(e) {
        e.preventDefault();
        const selectedCategory = $(this).data('category');
        activeFilters.category = selectedCategory.toLowerCase();
        
        // Visually indicate which category is selected (optional)
        $('.categories-btn').text('Category: ' + $(this).text());
        $(this).parent().hide(); // Hide dropdown

        // Remove 'active' class from other general filter buttons if needed, or manage category as a separate filter
        // $filterButtons.not($('.categories-btn')).removeClass('active'); // Example: if category overrides other filters

        applyFiltersAndSearch();
    });


    // Clear Filters button
    $clearFiltersButton.on('click', function() {
        activeFilters = {};
        $filterButtons.removeClass('active');
        $('.categories-btn').text('Categories ▼'); // Reset category button text
        $searchCustomersInput.val('');
        $tbody.empty().append(originalRows); // Restore original rows
        $('#customerCount').text(`Customers ${originalRows.length}`);
        $(this).hide();
        // Reset sort indicators as well if desired
        $headers.data('direction', null).find('.sort-indicator').removeClass('asc desc').text('');
    });

    // Debounce function to limit rate of function execution (for search)
    function debounce(func, wait, immediate) {
        var timeout;
        return function() {
            var context = this, args = arguments;
            var later = function() {
                timeout = null;
                if (!immediate) func.apply(context, args);
            };
            var callNow = immediate && !timeout;
            clearTimeout(timeout);
            timeout = setTimeout(later, wait);
            if (callNow) func.apply(context, args);
        };
    };

    // Close dropdown if clicked outside
    $(window).on('click', function(event) {
        if (!$(event.target).closest('.dropdown').length) {
            $('.dropdown-content').hide();
        }
    });
});
```

**JavaScript Explanation:**

* **Sorting:**
    * It's client-side: reads the table, sorts the rows based on text content, and re-appends them. Good for small to medium datasets. For large datasets, server-side sorting is preferred.
    * Updates the sort indicators (▲/▼).
* **Delete Confirmation:**
    * Shows a `confirm()` dialog.
    * If confirmed, it creates a temporary form and submits it via `POST` to the delete URL. This is a common way to trigger a `POST` request for deletion without AJAX, and allows CSRF protection if your Flask app uses it. (Make sure your delete route expects `POST`).
* **Filter & Search:**
    * Combines search and filter logic.
    * `applyFiltersAndSearch` function re-evaluates all visible rows based on the search term and active filters.
    * `debounce` is used on the search input to prevent too many rapid executions.
    * "Clear All Filters" button resets everything.
    * The categories filter is a basic dropdown example; you might have a more sophisticated multi-select or similar.
    * **Note on `data-has-price-list`**: For the "With Price-list" filter to work as in the JS example, you'd need to ensure your Flask template adds this data attribute to the `<tr>` elements if a customer has a price list. E.g., `<tr data-has-price-list="{{ 'true' if customer.has_price_list else 'false' }}">`.
