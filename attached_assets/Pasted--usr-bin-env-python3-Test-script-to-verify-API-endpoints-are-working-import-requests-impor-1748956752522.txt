#!/usr/bin/env python3
"""
Test script to verify API endpoints are working
"""
import requests
import json
import sys

def test_api_endpoint(base_url, endpoint, params=None):
    """Test a specific API endpoint"""
    url = f"{base_url}{endpoint}"
    
    try:
        print(f"\n--- Testing: {url} ---")
        if params:
            print(f"Parameters: {params}")
            
        response = requests.get(url, params=params, timeout=10)
        
        print(f"Status Code: {response.status_code}")
        print(f"Headers: {dict(response.headers)}")
        
        if response.status_code == 200:
            try:
                data = response.json()
                print(f"Response Data: {json.dumps(data, indent=2)}")
                return True
            except json.JSONDecodeError:
                print(f"Response Text: {response.text}")
                return False
        else:
            print(f"Error Response: {response.text}")
            return False
            
    except requests.exceptions.RequestException as e:
        print(f"Request failed: {e}")
        return False

def test_form_submission(base_url):
    """Test form submission"""
    url = f"{base_url}/orders/new"
    
    try:
        print(f"\n--- Testing Form GET: {url} ---")
        response = requests.get(url, timeout=10)
        print(f"Status Code: {response.status_code}")
        
        if response.status_code == 200:
            print("✓ Form page loads successfully")
            return True
        else:
            print(f"✗ Form page failed: {response.text}")
            return False
            
    except requests.exceptions.RequestException as e:
        print(f"Form test failed: {e}")
        return False

def main():
    # Get base URL
    if len(sys.argv) > 1:
        base_url = sys.argv[1].rstrip('/')
    else:
        base_url = "http://localhost:5000"
    
    print(f"Testing Flask API endpoints at: {base_url}")
    
    # Test basic connectivity
    try:
        response = requests.get(base_url, timeout=5)
        print(f"✓ Server is responding (Status: {response.status_code})")
    except:
        print("✗ Cannot connect to server")
        return
    
    # Test API endpoints
    tests = [
        ("/api/products/search", {"q": "test"}),
        ("/api/products/search", {"q": "a"}),
        ("/api/products/search", {}),
    ]
    
    results = []
    
    for endpoint, params in tests:
        result = test_api_endpoint(base_url, endpoint, params)
        results.append((endpoint, params, result))
    
    # Test form page
    form_result = test_form_submission(base_url)
    
    # Summary
    print("\n" + "="*50)
    print("TEST SUMMARY")
    print("="*50)
    
    for endpoint, params, result in results:
        status = "✓ PASS" if result else "✗ FAIL"
        print(f"{status}: {endpoint} {params}")
    
    print(f"{'✓ PASS' if form_result else '✗ FAIL'}: Form page load")
    
    # Recommendations
    print("\n" + "="*50)
    print("RECOMMENDATIONS")
    print("="*50)
    
    if not any(result for _, _, result in results):
        print("1. Check if your Flask routes are properly defined")
        print("2. Verify your database has product data")
        print("3. Check Flask app is running in debug mode")
        print("4. Review server logs for errors")
    
    if not form_result:
        print("1. Check if templates exist and are correctly named")
        print("2. Verify form imports and model definitions")
        print("3. Check for template rendering errors")

if __name__ == "__main__":
    main()
