## 3  Nightly *Daily Orders* job (auto-convert quotations ➜ orders)

**Service** (`services/daily_orders.py`):

```python
from datetime import date
from sqlalchemy.exc import IntegrityError
from models import db, Quotation, Order, OrderStatus

def generate_daily_orders(run_date: date | None = None):
    run_date = run_date or date.today()
    accepted_quotes = (Quotation.query
                       .filter(Quotation.status == 'accepted',
                               db.func.date(Quotation.accepted_at) == run_date)
                       .all())
    for q in accepted_quotes:
        if q.order:          # already converted – idempotent
            continue
        order = Order(quotation_id=q.id,
                      order_number=f"O{q.id:06d}",
                      status=OrderStatus.pending)
        db.session.add(order)
        q.status = 'converted'
    try:
        db.session.commit()
    except IntegrityError:
        db.session.rollback()
        # log & continue – the unique quotation_id index keeps us safe
```

**Scheduler** (simple in-process):

```python
from apscheduler.schedulers.background import BackgroundScheduler
from pytz import timezone

def schedule(app):
    sched = BackgroundScheduler(timezone=timezone('Europe/Nicosia'))
    sched.add_job(lambda: generate_daily_orders(),
                  'cron', hour=2, minute=0, id='daily_orders')
    sched.start()
```

Call `schedule(app)` once in `app.py` **after** the application context is ready.
