### Alembic migration script
```python
def upgrade():
    op.add_column('quotation_item', sa.Column('position', sa.Integer(), nullable=False, server_default='0'))

    conn = op.get_bind()
    result = conn.execute(sa.text("SELECT id FROM quotation"))
    for row in result:
        items = conn.execute(sa.text(f"""
            SELECT id FROM quotation_item WHERE quotation_id = :qid ORDER BY id
        """), {'qid': row.id}).fetchall()

        for idx, item in enumerate(items):
            conn.execute(sa.text("""
                UPDATE quotation_item SET position = :pos WHERE id = :id
            """), {'pos': idx, 'id': item.id})
```

<script>
  // Reorder items via SortableJS
  new Sortable(document.getElementById('quotation-items'), {
    onEnd: function () {
      const order = Array.from(document.querySelectorAll('#quotation-items li')).map(li => li.dataset.id);
      fetch('{{ url_for("reorder_quotation_items", quotation_id=quotation.id) }}', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ order: order })
      });
    }
  });

  // Add new supplier
  function submitNewSupplier() {
    const name = document.getElementById('new_supplier_name').value;
    fetch('{{ url_for("add_supplier") }}', {
      method: 'POST',
      headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
      body: new URLSearchParams({ name })
    }).then(res => res.json()).then(data => {
      if (data.id) {
        document.querySelectorAll('select[name="supplier_id"]').forEach(select => {
          const option = new Option(data.name, data.id);
          select.add(option);
          select.value = data.id;
        });
        const modal = bootstrap.Modal.getInstance(document.getElementById('supplierModal'));
        modal.hide();
      } else {
        alert(data.error);
      }
    });
  }

  // Add new item dynamically
  function addItem() {
    const desc = document.getElementById('new_item_description').value;
    if (!desc) return alert("Description required");
    fetch('{{ url_for("add_quotation_item", quotation_id=quotation.id) }}', {
      method: 'POST',
      headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
      body: new URLSearchParams({ description: desc })
    }).then(res => res.json()).then(data => {
      location.reload();  // reload to see new item
    });
  }
</script>
