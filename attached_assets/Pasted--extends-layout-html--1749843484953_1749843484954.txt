{% block scripts %}
  <script>
    const master = document.getElementById('master-cb');
    const body   = document.getElementById('pl-body');
    const bulk   = document.getElementById('bulk-bar');
    const rows   = () => Array.from(document.querySelectorAll('.row-cb'));

    const updateBulk = () => {
      const any = rows().some(cb => cb.checked);
      bulk.classList.toggle('show', any);
    };

    master?.addEventListener('change', e => {
      rows().forEach(cb => cb.checked = e.target.checked);
      updateBulk();
    });

    body?.addEventListener('change', e => {
      if(e.target.classList.contains('row-cb')) updateBulk();
    });

    document.getElementById('select-all')?.addEventListener('click', () => {
      rows().forEach(cb => cb.checked = true); master.checked = true; updateBulk();
    });
    document.getElementById('deselect-all')?.addEventListener('click', () => {
      rows().forEach(cb => cb.checked = false); master.checked = false; updateBulk();
    });

    document.getElementById('bulk-delete')?.addEventListener('click', () => {
      const ids = rows().filter(cb => cb.checked).map(cb => cb.value);
      if(!ids.length) return;
      if(confirm(`Delete ${ids.length} price entries?`)){
        fetch('/price-lists/batch-delete',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({price_ids:ids})})
          .then(r=>r.ok?location.reload():alert('Error'));
      }
    });

    function applyFilter(){
      const c = document.getElementById('customer-filter').value;
      const k = document.getElementById('category-filter').value;
      const searchParams = new URLSearchParams(window.location.search);
      c ? searchParams.set('customer_id', c) : searchParams.delete('customer_id');
      k ? searchParams.set('category', k) : searchParams.delete('category');
      window.location.search = searchParams.toString();
    }
  </script>
{% endblock %}
