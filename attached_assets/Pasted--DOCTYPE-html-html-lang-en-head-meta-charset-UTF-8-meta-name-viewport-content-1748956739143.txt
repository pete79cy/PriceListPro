    <script>
        // Debug and working JavaScript for order form
        
        let orderItems = [];
        let itemCounter = 0;

        // Debug function
        function debugLog(message, data = null) {
            console.log(`[DEBUG] ${message}`, data);
            
            // Show debug info on page
            const debugEl = document.getElementById('search-debug') || document.getElementById('manual-debug');
            if (debugEl) {
                debugEl.style.display = 'block';
                debugEl.innerHTML += `<div><strong>${new Date().toLocaleTimeString()}:</strong> ${message}</div>`;
                if (data) {
                    debugEl.innerHTML += `<pre>${JSON.stringify(data, null, 2)}</pre>`;
                }
            }
        }

        // Test product search functionality
        async function testProductSearch(query) {
            debugLog(`Testing product search with query: "${query}"`);
            
            try {
                const response = await fetch(`/api/products/search?q=${encodeURIComponent(query)}`);
                debugLog(`Search response status: ${response.status}`);
                
                if (!response.ok) {
                    throw new Error(`HTTP ${response.status}: ${response.statusText}`);
                }
                
                const products = await response.json();
                debugLog(`Found ${products.length} products`, products);
                
                // Display results
                const resultsEl = document.getElementById('search-results');
                if (products.length > 0) {
                    resultsEl.innerHTML = products.map(p => 
                        `<div class="border p-2 mb-1">
                            <strong>${p.name}</strong> - €${p.price} per ${p.unit}
                            <button class="btn btn-sm btn-primary ms-2" onclick="addProductToOrder(${p.id}, '${p.name}', ${p.price}, '${p.unit}')">
                                Add to Order
                            </button>
                        </div>`
                    ).join('');
                } else {
                    resultsEl.innerHTML = '<div class="text-muted">No products found</div>';
                }
                
            } catch (error) {
                debugLog(`Search error: ${error.message}`);
                document.getElementById('search-results').innerHTML = 
                    `<div class="alert alert-danger">Search failed: ${error.message}</div>`;
            }
        }

        // Add product to order
        function addProductToOrder(productId, productName, price, unit) {
            debugLog(`Adding product to order: ${productName}`);
            
            const item = {
                id: ++itemCounter,
                product_id: productId,
                product_name: productName,
                quantity: 1,
                unit_price: price,
                unit: unit,
                vat_rate: 19,
                line_total: price
            };
            
            orderItems.push(item);
            renderOrderItems();
            calculateTotals();
        }

        // Add manual item
        function addManualItem() {
            debugLog('Adding manual item');
            
            const item = {
                id: ++itemCounter,
                product_id: null,
                product_name: 'Manual Item',
                quantity: 1,
                unit_price: 0,
                unit: 'pcs',
                vat_rate: 19,
                line_total: 0
            };
            
            orderItems.push(item);
            renderOrderItems();
        }

        // Render order items
        function renderOrderItems() {
            const container = document.getElementById('order-items-container');
            
            if (orderItems.length === 0) {
                container.innerHTML = '<p class="text-muted">No items added yet. Use the search above or add manual items.</p>';
                return;
            }
            
            container.innerHTML = orderItems.map(item => `
                <div class="card mb-2" data-item-id="${item.id}">
                    <div class="card-body">
                        <div class="row align-items-center">
                            <div class="col-md-3">
                                <input type="text" class="form-control form-control-sm" 
                                       value="${item.product_name}" 
                                       onchange="updateItemName(${item.id}, this.value)">
                            </div>
                            <div class="col-md-2">
                                <input type="number" class="form-control form-control-sm" 
                                       value="${item.quantity}" step="0.001" min="0.001"
                                       onchange="updateItemQuantity(${item.id}, this.value)">
                            </div>
                            <div class="col-md-2">
                                <div class="input-group input-group-sm">
                                    <span class="input-group-text">€</span>
                                    <input type="number" class="form-control" 
                                           value="${item.unit_price}" step="0.01" min="0"
                                           onchange="updateItemPrice(${item.id}, this.value)">
                                </div>
                            </div>
                            <div class="col-md-2">
                                <select class="form-select form-select-sm" 
                                        onchange="updateItemVAT(${item.id}, this.value)">
                                    <option value="0" ${item.vat_rate === 0 ? 'selected' : ''}>0%</option>
                                    <option value="5" ${item.vat_rate === 5 ? 'selected' : ''}>5%</option>
                                    <option value="19" ${item.vat_rate === 19 ? 'selected' : ''}>19%</option>
                                </select>
                            </div>
                            <div class="col-md-2">
                                <strong>€${item.line_total.toFixed(2)}</strong>
                            </div>
                            <div class="col-md-1">
                                <button type="button" class="btn btn-sm btn-outline-danger" 
                                        onclick="removeItem(${item.id})">
                                    ×
                                </button>
                            </div>
                        </div>
                    </div>
                </div>
            `).join('');
        }

        // Update item functions
        function updateItemName(itemId, name) {
            const item = orderItems.find(i => i.id === itemId);
            if (item) {
                item.product_name = name;
                debugLog(`Updated item ${itemId} name to: ${name}`);
            }
        }

        function updateItemQuantity(itemId, quantity) {
            const item = orderItems.find(i => i.id === itemId);
            if (item) {
                item.quantity = parseFloat(quantity) || 0;
                item.line_total = item.quantity * item.unit_price;
                renderOrderItems();
                calculateTotals();
                debugLog(`Updated item ${itemId} quantity to: ${quantity}`);
            }
        }

        function updateItemPrice(itemId, price) {
            const item = orderItems.find(i => i.id === itemId);
            if (item) {
                item.unit_price = parseFloat(price) || 0;
                item.line_total = item.quantity * item.unit_price;
                renderOrderItems();
                calculateTotals();
                debugLog(`Updated item ${itemId} price to: ${price}`);
            }
        }

        function updateItemVAT(itemId, vatRate) {
            const item = orderItems.find(i => i.id === itemId);
            if (item) {
                item.vat_rate = parseInt(vatRate) || 0;
                calculateTotals();
                debugLog(`Updated item ${itemId} VAT to: ${vatRate}%`);
            }
        }

        function removeItem(itemId) {
            orderItems = orderItems.filter(i => i.id !== itemId);
            renderOrderItems();
            calculateTotals();
            debugLog(`Removed item ${itemId}`);
        }

        // Calculate totals
        function calculateTotals() {
            const subtotal = orderItems.reduce((sum, item) => sum + item.line_total, 0);
            const vatAmount = orderItems.reduce((sum, item) => {
                return sum + (item.line_total * item.vat_rate / 100);
            }, 0);
            const total = subtotal + vatAmount;

            document.getElementById('subtotal').textContent = `€${subtotal.toFixed(2)}`;
            document.getElementById('vat-amount').textContent = `€${vatAmount.toFixed(2)}`;
            document.getElementById('total').textContent = `€${total.toFixed(2)}`;
        }

        // Event listeners
        document.addEventListener('DOMContentLoaded', function() {
            debugLog('DOM loaded, setting up event listeners');

            // Test search button
            document.getElementById('test-search-btn')?.addEventListener('click', function() {
                const query = document.getElementById('product-search').value;
                testProductSearch(query);
            });

            // Add manual button test
            document.getElementById('add-manual-btn')?.addEventListener('click', function() {
                addManualItem();
                document.getElementById('manual-debug').style.display = 'block';
                document.getElementById('manual-debug').innerHTML = '<div>Manual item added successfully!</div>';
            });

            // Real product search
            const searchInput = document.getElementById('product-search-input');
            const searchBtn = document.getElementById('search-btn');
            
            if (searchInput && searchBtn) {
                // Search on button click
                searchBtn.addEventListener('click', function() {
                    const query = searchInput.value.trim();
                    if (query.length >= 2) {
                        performProductSearch(query);
                    }
                });

                // Search on Enter key
                searchInput.addEventListener('keypress', function(e) {
                    if (e.key === 'Enter') {
                        e.preventDefault();
                        const query = this.value.trim();
                        if (query.length >= 2) {
                            performProductSearch(query);
                        }
                    }
                });

                // Search as you type (with debounce)
                let searchTimeout;
                searchInput.addEventListener('input', function() {
                    clearTimeout(searchTimeout);
                    const query = this.value.trim();
                    
                    if (query.length >= 2) {
                        searchTimeout = setTimeout(() => {
                            performProductSearch(query);
                        }, 300);
                    } else {
                        document.getElementById('product-search-results').style.display = 'none';
                    }
                });
            }

            // Add manual item button
            document.getElementById('add-manual-item')?.addEventListener('click', addManualItem);

            // Set today's date
            const dateInput = document.getElementById('order-date');
            if (dateInput) {
                dateInput.value = new Date().toISOString().split('T')[0];
            }

            debugLog('Event listeners set up successfully');
        });

        // Perform product search
        async function performProductSearch(query) {
            debugLog(`Performing product search: ${query}`);
            
            try {
                const response = await fetch(`/api/products/search?q=${encodeURIComponent(query)}`);
                
                if (!response.ok) {
                    throw new Error(`HTTP ${response.status}`);
                }
                
                const products = await response.json();
                displaySearchResults(products);
                
            } catch (error) {
                debugLog(`Search error: ${error.message}`);
                const resultsEl = document.getElementById('product-search-results');
                resultsEl.innerHTML = `<div class="list-group-item text-danger">Search failed: ${error.message}</div>`;
                resultsEl.style.display = 'block';
            }
        }

        // Display search results
        function displaySearchResults(products) {
            const resultsEl = document.getElementById('product-search-results');
            
            if (products.length === 0) {
                resultsEl.innerHTML = '<div class="list-group-item text-muted">No products found</div>';
            } else {
                resultsEl.innerHTML = products.map(product => `
                    <button type="button" class="list-group-item list-group-item-action" 
                            onclick="addProductToOrder(${product.id}, '${product.name.replace(/'/g, "\\'")}', ${product.price}, '${product.unit}')">
                        <div class="d-flex w-100 justify-content-between">
                            <h6 class="mb-1">${product.name}</h6>
                            <small>€${product.price} per ${product.unit}</small>
                        </div>
                    </button>
                `).join('');
            }
            
            resultsEl.style.display = 'block';
        }
    </script>
