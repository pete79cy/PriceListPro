Here's an enhanced implementation of the `generate_custom_supplier_report()` function in `utils/pdf_generator.py` using **FPDF (Python library)**, integrating several of the suggested improvements:

---

### ✅ Included Enhancements:

- Company Header  
- Cover Page with summary  
- Per-supplier grouping and subtotals  
- Unit price and line total  
- Notes section  
- Page numbers  
- Watermark ("Confidential")  
- Table styling with alternating rows  
- Optional terms and validity date

---

### 🧾 `generate_custom_supplier_report` Implementation

```python
from fpdf import FPDF
from flask import current_app
from datetime import datetime, timedelta
import io
from models import CompanySettings
import logging

logger = logging.getLogger(__name__)

class PDF(FPDF):
    def header(self):
        if self.company:
            self.set_font("Helvetica", 'B', 12)
            self.cell(0, 10, self.company.name, ln=True)
            self.set_font("Helvetica", '', 10)
            self.cell(0, 5, self.company.address or '', ln=True)
            self.cell(0, 5, self.company.email or '', ln=True)
            self.ln(5)

    def footer(self):
        self.set_y(-15)
        self.set_font("Helvetica", 'I', 8)
        self.set_text_color(128)
        self.cell(0, 10, f"Page {self.page_no()} / {{nb}}", align='C')

    def add_watermark(self, text):
        self.set_font("Helvetica", 'B', 40)
        self.set_text_color(240, 240, 240)
        self.rotate(45, x=self.w / 3, y=self.h / 2)
        self.text(self.w / 3, self.h / 2, text)
        self.rotate(0)

def generate_custom_supplier_report(quotation, selected_suppliers, selected_fields,
                                    include_prices=True, include_company_header=True,
                                    include_terms=True, group_by_supplier=True,
                                    notes=None):
    try:
        pdf = PDF()
        pdf.alias_nb_pages()
        pdf.add_page()

        # Company settings
        company = CompanySettings.query.first()
        pdf.company = company if include_company_header else None

        # Cover page content
        pdf.set_font("Helvetica", 'B', 16)
        pdf.cell(0, 10, f"Quotation Report: {quotation.quotation_number}", ln=True)
        pdf.set_font("Helvetica", '', 12)
        pdf.cell(0, 10, f"Date Issued: {quotation.date.strftime('%Y-%m-%d')}", ln=True)
        pdf.cell(0, 10, f"Valid Until: {(quotation.valid_until or (quotation.date + timedelta(days=30))).strftime('%Y-%m-%d')}", ln=True)
        pdf.cell(0, 10, f"Customer: {quotation.customer.name if quotation.customer else 'N/A'}", ln=True)
        pdf.cell(0, 10, f"Suppliers Included: {', '.join(selected_suppliers)}", ln=True)
        pdf.ln(10)

        # Add watermark
        pdf.add_watermark("CONFIDENTIAL")

        # Grouped supplier itemization
        for supplier_name in selected_suppliers:
            items = [i for i in quotation.items if i.supplier_name == supplier_name]
            if not items:
                continue

            pdf.set_font("Helvetica", 'B', 14)
            pdf.set_text_color(0)
            pdf.cell(0, 10, f"Supplier: {supplier_name}", ln=True)
            pdf.set_font("Helvetica", 'B', 10)

            # Table headers
            headers = ["Item", "Qty"]
            if include_prices:
                headers.extend(["Unit Price", "Total"])
            for header in headers:
                pdf.cell(40, 8, header, border=1)
            pdf.ln()

            # Table rows
            pdf.set_font("Helvetica", '', 10)
            total = 0
            fill = False
            for item in items:
                pdf.set_fill_color(245 if fill else 255)
                pdf.cell(40, 8, item.product_name, border=1, fill=1)
                pdf.cell(40, 8, str(item.quantity), border=1, fill=1)
                if include_prices:
                    pdf.cell(40, 8, f"€{item.unit_price:.2f}", border=1, fill=1)
                    line_total = item.unit_price * item.quantity
                    total += line_total
                    pdf.cell(40, 8, f"€{line_total:.2f}", border=1, fill=1)
                pdf.ln()
                fill = not fill

            if include_prices:
                pdf.set_font("Helvetica", 'B', 10)
                pdf.cell(120, 8, "Subtotal", border=1)
                pdf.cell(40, 8, f"€{total:.2f}", border=1)
                pdf.ln(15)

        # Notes section
        if notes:
            pdf.set_font("Helvetica", 'I', 10)
            pdf.set_text_color(50)
            pdf.multi_cell(0, 10, f"Notes: {notes}", border=1)
            pdf.ln()

        # Terms and Conditions
        if include_terms:
            pdf.add_page()
            pdf.set_font("Helvetica", 'B', 12)
            pdf.cell(0, 10, "Terms and Conditions", ln=True)
            pdf.set_font("Helvetica", '', 10)
            pdf.multi_cell(0, 6, company.terms if company and company.terms else "Standard terms apply.")
            pdf.ln()

        # Final output
        pdf_output = io.BytesIO()
        pdf.output(pdf_output)
        pdf_output.seek(0)
        filename = f"quotation_report_{quotation.quotation_number}.pdf"
        return pdf_output.read(), filename

    except Exception as e:
        logger.error(f"Error generating custom supplier report: {str(e)}")
        raise
```
