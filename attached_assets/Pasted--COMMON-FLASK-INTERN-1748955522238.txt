from flask import Flask
from flask_sqlalchemy import SQLAlchemy
from flask_admin import Admin
from flask_admin.contrib.sqla import ModelView

# Initialize extensions
db = SQLAlchemy()
admin = Admin()

def create_app():
    app = Flask(__name__)
    app.config['SECRET_KEY'] = 'your-secret-key'
    app.config['SQLALCHEMY_DATABASE_URI'] = 'your-database-uri'
    app.config['SQLALCHEMY_TRACK_MODIFICATIONS'] = False
    
    # Initialize extensions with app
    db.init_app(app)
    admin.init_app(app)
    
    # Import models AFTER db is initialized
    from models import Order, Customer, Product, OrderItem
    
    # Add admin views
    admin.add_view(ModelView(Customer, db.session))
    admin.add_view(ModelView(Product, db.session))
    admin.add_view(ModelView(Order, db.session))
    
    # Register blueprints
    from routes import main_bp
    app.register_blueprint(main_bp)
    
    return app

from flask_sqlalchemy import SQLAlchemy
from datetime import datetime
from enum import Enum

# Import db from app instead of creating new instance
from app import db

class OrderStatus(Enum):
    PENDING = "pending"
    CONFIRMED = "confirmed"
    DELIVERED = "delivered"
    CANCELLED = "cancelled"

class Customer(db.Model):
    __tablename__ = 'customers'
    
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(100), nullable=False)
    email = db.Column(db.String(120), unique=True, nullable=True)
    phone = db.Column(db.String(20), nullable=True)
    address = db.Column(db.Text, nullable=True)
    created_at = db.Column(db.DateTime, default=datetime.utcnow)
    
    # Relationships
    orders = db.relationship('Order', backref='customer', lazy=True)
    price_lists = db.relationship('CustomerPriceList', backref='customer', lazy=True)

class Product(db.Model):
    __tablename__ = 'products'
    
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(200), nullable=False)
    description = db.Column(db.Text, nullable=True)
    base_price = db.Column(db.Numeric(10, 2), nullable=False, default=0.00)
    unit = db.Column(db.String(20), nullable=True, default='pcs')
    category = db.Column(db.String(50), nullable=True)
    created_at = db.Column(db.DateTime, default=datetime.utcnow)
    
    # Relationships
    order_items = db.relationship('OrderItem', backref='product', lazy=True)
    customer_prices = db.relationship('CustomerPriceList', backref='product', lazy=True)

class Order(db.Model):
    __tablename__ = 'orders'
    
    id = db.Column(db.Integer, primary_key=True)
    customer_id = db.Column(db.Integer, db.ForeignKey('customers.id'), nullable=False)
    order_date = db.Column(db.Date, nullable=False, default=datetime.utcnow().date)
    status = db.Column(db.Enum(OrderStatus), nullable=False, default=OrderStatus.PENDING)
    notes = db.Column(db.Text, nullable=True)
    subtotal = db.Column(db.Numeric(10, 2), nullable=False, default=0.00)
    vat_rate = db.Column(db.Numeric(5, 2), nullable=False, default=19.00)
    vat_amount = db.Column(db.Numeric(10, 2), nullable=False, default=0.00)
    total = db.Column(db.Numeric(10, 2), nullable=False, default=0.00)
    created_at = db.Column(db.DateTime, default=datetime.utcnow)
    
    # Relationships
    items = db.relationship('OrderItem', backref='order', lazy=True, cascade='all, delete-orphan')

class OrderItem(db.Model):
    __tablename__ = 'order_items'
    
    id = db.Column(db.Integer, primary_key=True)
    order_id = db.Column(db.Integer, db.ForeignKey('orders.id'), nullable=False)
    product_id = db.Column(db.Integer, db.ForeignKey('products.id'), nullable=False)
    quantity = db.Column(db.Numeric(10, 3), nullable=False, default=1.000)
    unit_price = db.Column(db.Numeric(10, 2), nullable=False, default=0.00)
    vat_rate = db.Column(db.Numeric(5, 2), nullable=False, default=19.00)
    line_total = db.Column(db.Numeric(10, 2), nullable=False, default=0.00)

class CustomerPriceList(db.Model):
    __tablename__ = 'customer_price_lists'
    
    id = db.Column(db.Integer, primary_key=True)
    customer_id = db.Column(db.Integer, db.ForeignKey('customers.id'), nullable=False)
    product_id = db.Column(db.Integer, db.ForeignKey('products.id'), nullable=False)
    price = db.Column(db.Numeric(10, 2), nullable=False)
    created_at = db.Column(db.DateTime, default=datetime.utcnow)
    
    # Unique constraint
    __table_args__ = (db.UniqueConstraint('customer_id', 'product_id'),)

from flask_wtf import FlaskForm
from wtforms import StringField, SelectField, TextAreaField, DecimalField, IntegerField, DateField, FieldList, FormField
from wtforms.validators import DataRequired, NumberRange, Optional
from wtforms.ext.sqlalchemy.fields import QuerySelectField
from datetime import date

# Import models properly
from models import Customer, Product, OrderStatus

def get_customers():
    """Helper function for QuerySelectField"""
    try:
        return Customer.query.all()
    except:
        return []

def get_products():
    """Helper function for QuerySelectField"""
    try:
        return Product.query.all()
    except:
        return []

class OrderItemForm(FlaskForm):
    product_id = QuerySelectField(
        'Product',
        query_factory=get_products,
        get_label='name',
        validators=[DataRequired()],
        allow_blank=False
    )
    quantity = DecimalField(
        'Quantity',
        validators=[DataRequired(), NumberRange(min=0.001)],
        default=1.000,
        places=3
    )
    unit_price = DecimalField(
        'Unit Price',
        validators=[DataRequired(), NumberRange(min=0)],
        default=0.00,
        places=2
    )
    vat_rate = SelectField(
        'VAT Rate',
        choices=[('0', '0%'), ('5', '5%'), ('19', '19%')],
        default='19',
        validators=[DataRequired()]
    )

class OrderForm(FlaskForm):
    customer_id = QuerySelectField(
        'Customer',
        query_factory=get_customers,
        get_label='name',
        validators=[DataRequired()],
        allow_blank=False
    )
    order_date = DateField(
        'Order Date',
        validators=[DataRequired()],
        default=date.today
    )
    status = SelectField(
        'Status',
        choices=[(status.value, status.value.title()) for status in OrderStatus],
        default=OrderStatus.PENDING.value,
        validators=[DataRequired()]
    )
    notes = TextAreaField('Notes', validators=[Optional()])
    items = FieldList(FormField(OrderItemForm), min_entries=1)

from flask import Blueprint, render_template, request, redirect, url_for, flash, jsonify
from models import db, Order, Customer, Product, OrderItem, CustomerPriceList, OrderStatus
from forms import OrderForm
from datetime import datetime
import json

main_bp = Blueprint('main', __name__)

@main_bp.route('/')
def index():
    return render_template('index.html')

@main_bp.route('/orders')
def orders():
    try:
        orders = Order.query.order_by(Order.created_at.desc()).all()
        return render_template('orders.html', orders=orders)
    except Exception as e:
        flash(f'Error loading orders: {str(e)}', 'error')
        return render_template('orders.html', orders=[])

@main_bp.route('/orders/new', methods=['GET', 'POST'])
def new_order():
    try:
        form = OrderForm()
        
        if form.validate_on_submit():
            # Create new order
            order = Order(
                customer_id=form.customer_id.data.id,
                order_date=form.order_date.data,
                status=OrderStatus(form.status.data),
                notes=form.notes.data
            )
            
            db.session.add(order)
            db.session.flush()  # Get order ID
            
            # Add order items
            subtotal = 0
            for item_form in form.items:
                if item_form.product_id.data:
                    item = OrderItem(
                        order_id=order.id,
                        product_id=item_form.product_id.data.id,
                        quantity=item_form.quantity.data,
                        unit_price=item_form.unit_price.data,
                        vat_rate=float(item_form.vat_rate.data)
                    )
                    item.line_total = item.quantity * item.unit_price
                    subtotal += item.line_total
                    db.session.add(item)
            
            # Calculate totals
            order.subtotal = subtotal
            order.vat_amount = subtotal * (order.items[0].vat_rate / 100) if order.items else 0
            order.total = order.subtotal + order.vat_amount
            
            db.session.commit()
            flash('Order created successfully!', 'success')
            return redirect(url_for('main.orders'))
        
        return render_template('new_order.html', form=form)
        
    except Exception as e:
        db.session.rollback()
        flash(f'Error creating order: {str(e)}', 'error')
        return render_template('new_order.html', form=OrderForm())

@main_bp.route('/api/products/search')
def search_products():
    """API endpoint for product search"""
    try:
        query = request.args.get('q', '')
        if len(query) < 2:
            return jsonify([])
        
        products = Product.query.filter(
            Product.name.contains(query)
        ).limit(10).all()
        
        return jsonify([{
            'id': p.id,
            'name': p.name,
            'price': float(p.base_price),
            'unit': p.unit
        } for p in products])
        
    except Exception as e:
        return jsonify({'error': str(e)}), 500

@main_bp.route('/api/customer/<int:customer_id>/product/<int:product_id>/price')
def get_customer_product_price():
    """Get customer-specific price for a product"""
    try:
        customer_id = request.view_args['customer_id']
        product_id = request.view_args['product_id']
        
        # Check for customer-specific price
        custom_price = CustomerPriceList.query.filter_by(
            customer_id=customer_id,
            product_id=product_id
        ).first()
        
        if custom_price:
            price = float(custom_price.price)
        else:
            # Use base product price
            product = Product.query.get(product_id)
            price = float(product.base_price) if product else 0.00
        
        return jsonify({'price': price})
        
    except Exception as e:
        return jsonify({'error': str(e)}), 500
