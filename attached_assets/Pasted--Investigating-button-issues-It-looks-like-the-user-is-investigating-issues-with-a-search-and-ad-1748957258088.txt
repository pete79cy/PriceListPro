### 2. Drop-in patch (replace the previous helpers or append)

```html
<script>
// ---------------------------------------------------------------------------
//  CENTRAL CONFIG (edit here only)
// ---------------------------------------------------------------------------
const API_BASE = '/orders/api';          // <— adjust once if your prefix changes
const SEARCH_ENDPOINT = `${API_BASE}/search/products`;

// ---------------------------------------------------------------------------
//  Override / extend the earlier helpers
// ---------------------------------------------------------------------------
async function testProductSearch(query) {
    debugLog(`Testing product search API with query: “${query}”`);
    try {
        const url = `${SEARCH_ENDPOINT}?q=${encodeURIComponent(query)}`;
        debugLog(`Requesting: ${url}`);

        const response = await fetch(url);
        debugLog(`Response status: ${response.status} ${response.statusText}`);

        if (!response.ok) {
            throw new Error(`HTTP ${response.status}: ${response.statusText}`);
        }
        const products = await response.json();
        debugLog(`Search successful, found ${products.length} products`, products);
        displaySearchResults(products);
    } catch (err) {
        debugLog(`Search failed: ${err.message}`);
        const resultsEl = document.getElementById('product-search-results');
        if (resultsEl) {
            resultsEl.innerHTML = `
                <div class="list-group-item text-danger">
                    Search failed: ${err.message}
                </div>`;
            resultsEl.style.display = 'block';
        }
    }
}

//  Enhanced manual-item stub so you know what went wrong
function addProductToOrder(id, name, price, unit) {
    const container = document.getElementById('order-items-container');
    if (!container) {
        debugLog('✗ order-items-container not found - cannot append item');
        alert('Developer note: Missing #order-items-container – check template.');
        return;
    }
    const row = document.createElement('div');
    row.className = 'list-group-item';
    row.innerHTML = `
        <div class="d-flex w-100 justify-content-between align-items-center">
            <div>
                <h6 class="my-0">${name}</h6>
                <small class="text-muted">${unit || ''}</small>
            </div>
            <span>€${Number(price).toFixed(2)}</span>
        </div>`;
    container.appendChild(row);
}

// Optional: if jQuery is required elsewhere, warn if missing
if (typeof jQuery === 'undefined') {
    debugLog('⚠️  jQuery is NOT loaded - some legacy code may fail');
}
</script>
```

**What changed?**

1. **Central `API_BASE` constant** – every future fetch can be built from that, so if the prefix ever changes again you won’t chase hard-coded strings.
2. **Search endpoint now matches back-end route** – `SEARCH_ENDPOINT` resolves to `/orders/api/search/products`.
3. **More defensive `addProductToOrder`** – fails loudly if the container is absent; great for template typos.
4. **jQuery presence check** – harmless log that immediately tells you if another script silently failed because jQuery wasn’t injected.

Load `/orders/enhanced/new`, open DevTools, and test again:

* **Search**: You should see a 200 OK to `/orders/api/search/products`.
* **Add manual item**: If the new row doesn’t appear, you’ll at least get a debug message or an alert telling you exactly why.
