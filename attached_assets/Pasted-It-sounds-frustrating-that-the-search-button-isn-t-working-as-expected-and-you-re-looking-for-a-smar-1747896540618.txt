```javascript
<script>
document.addEventListener('DOMContentLoaded', function () {
    // --- Configuration ---
    const productSearchInput = document.getElementById('productSearchInput');
    const productSearchButton = document.getElementById('productSearchButton');
    const productSelect = document.getElementById('productSelect');

    // --- "Smart" Search Function ---
    function performSmartSearch(query) {
        if (!productSelect) return;

        const originalQuery = query;
        query = query.toLowerCase().trim();
        productSelect.innerHTML = ''; // Clear existing options

        // Add the default "Choose a product" option
        const defaultOption = document.createElement('option');
        defaultOption.value = "";
        defaultOption.textContent = "Choose a product or search above";
        defaultOption.selected = true;
        defaultOption.disabled = true;
        productSelect.appendChild(defaultOption);

        if (!query && originalQuery.trim() === "") { // if query is empty after trim, show all or limited initial set
            // Optionally, populate with all products or a few popular ones if the query is empty
            // allProducts.slice(0, 20).forEach(product => addProductOption(product));
            // For now, we'll just leave it with the placeholder if search is empty.
            return;
        }
        
        if (allProducts.length === 0) {
            console.warn("No products loaded to search from.");
            const noProductOption = document.createElement('option');
            noProductOption.value = "";
            noProductOption.textContent = "Product list not available";
            noProductOption.disabled = true;
            productSelect.appendChild(noProductOption);
            return;
        }

        const searchTerms = query.split(/\s+/).filter(term => term.length > 0); // Split query into words

        const results = allProducts.map(product => {
            let score = 0;
            const productName = (product.name || "").toLowerCase();
            const productDescription = (product.description || "").toLowerCase();
            const productCode = (product.code || "").toLowerCase();
            const productSize = (product.size || "").toLowerCase();

            searchTerms.forEach(term => {
                if (productName.includes(term)) score += 10; // Higher score for name match
                if (productCode.includes(term)) score += 8;   // Score for code match
                if (productDescription.includes(term)) score += 3; // Lower score for description
                if (productSize.includes(term)) score += 2;   // Score for size match

                // Bonus for exact matches or starting with term
                if (productName.startsWith(term)) score += 5;
                if (productCode === term) score += 7;
            });

            return { product, score };
        })
        .filter(item => item.score > 0) // Keep only items with a score
        .sort((a, b) => b.score - a.score); // Sort by score, highest first

        if (results.length === 0) {
            const noResultsOption = document.createElement('option');
            noResultsOption.value = "";
            noResultsOption.textContent = "No products found matching your search.";
            noResultsOption.disabled = true;
            productSelect.appendChild(noResultsOption);
        } else {
            results.forEach(item => {
                addProductOption(item.product);
            });
        }
    }

    function addProductOption(product) {
        const option = document.createElement('option');
        option.value = product.id;
        // Display name, and optionally code or size for clarity
        option.textContent = `${product.name} (${product.code || 'N/A'}) ${product.size ? '- ' + product.size : ''}`;
        // You can store more data on the option if needed, e.g., using data-* attributes
        option.dataset.price = product.price; // Assuming product has a price property
        option.dataset.description = product.description;
        productSelect.appendChild(option);
    }

    // --- Event Listeners ---
    if (productSearchButton && productSearchInput) {
        productSearchButton.addEventListener('click', function () {
            performSmartSearch(productSearchInput.value);
        });

        productSearchInput.addEventListener('keyup', function (event) {
            // Search on Enter key or if text length is >= 3 (for real-time feedback)
            if (event.key === 'Enter') {
                performSmartSearch(productSearchInput.value);
            } else if (productSearchInput.value.length >= 2 || productSearchInput.value.length === 0) {
                // Trigger search for 2+ chars or when input is cleared
                performSmartSearch(productSearchInput.value);
            }
        });
    } else {

    }

        });
    }

    // --- Handle Auto-update Unit Price on Product Selection ---
```

**Explanation of the JavaScript:**
1.  **`allProducts` Array**: This is **CRITICAL**. You **must** ensure this array is populated with your product data. The example shows how you might parse it if it's embedded as JSON in your HTML by your server-side Flask/Jinja2 template, or you can hardcode it for testing.
2.  **`performSmartSearch(query)`**:
    * Clears previous options in the `productSelect` dropdown.
    * Converts the search `query` to lowercase for case-insensitive matching.
    * Splits the query into individual search terms (to match multiple words).
    * Iterates through `allProducts`:
        * Calculates a `score` for each product based on how well it matches the search terms across `name`, `description`, `code`, and `size`. Matches in `name` and `code` get higher scores. Starting matches also get a bonus.
        * Filters out products with a score of 0.
        * Sorts the results by `score` in descending order (best matches first).
    * Populates the `productSelect` dropdown with the filtered and sorted results. If no results, it shows a "No products found" message.
3.  **`addProductOption(product)`**: Helper function to create and add an `<option>` to the select dropdown.
4.  **Event Listeners**:
    * The `productSearchButton` listens for a `click` to trigger `performSmartSearch`.
    * The `productSearchInput` listens for `keyup`. It triggers the search on "Enter" or if the input length is 2 or more characters (providing real-time search). It also triggers if the input is cleared.
5.  **Modal `show.bs.modal` Event**: When the "Add Item to Order" modal is shown, it clears any previous search input and resets the product dropdown. It also sets focus to the search input.
6.  **Product Selection Change**: An optional event listener on `productSelect` demonstrates how you could potentially auto-fill the unit price if your `product` objects contain price information and you store it in `data-price` attributes on the options.
