    <script>
      // Checkbox selection → bulk bar
      const selectAll = document.getElementById('selectAll');
      const bulkBar   = document.getElementById('bulkBar');
      const rowBoxes  = Array.from(document.querySelectorAll('.row-checkbox'));
      const deleteBtn = document.getElementById('deleteSelected');

      const syncBulk = () => {
        const any = rowBoxes.some(cb => cb.checked);
        bulkBar.classList.toggle('show', any);
      };
      selectAll?.addEventListener('change', e => {
        rowBoxes.forEach(cb => cb.checked = e.target.checked);
        syncBulk();
      });
      rowBoxes.forEach(cb => cb.addEventListener('change', syncBulk));

      // Segmented control state
      document.querySelectorAll('#filterGroup button').forEach(btn => {
        btn.addEventListener('click', () => {
          document.querySelectorAll('#filterGroup button').forEach(b => b.classList.remove('active'));
          btn.classList.add('active');
        });
      });

      // Ellipsis contextual menu (placeholder)
      document.querySelectorAll('.ellipsis-btn').forEach(btn => {
        btn.addEventListener('click', () => {
          const id = btn.dataset.id;
          // TODO: Build contextual pop‑over using Popper / Tippy
          alert(`Quick actions for product #${id}`);
        });
      });

      // Delete selected placeholder
      deleteBtn?.addEventListener('click', () => {
        const ids = rowBoxes.filter(cb => cb.checked).map(cb => cb.value);
        if(!ids.length) return;
        if(confirm(`Delete ${ids.length} products?`)){
          fetch('/products/batch-delete',{
            method:'POST', headers:{'Content-Type':'application/json'}, body:JSON.stringify({product_ids:ids})
          }).then(res=>res.ok?location.reload():alert('Error deleting'));
        }
      });
    </script>
